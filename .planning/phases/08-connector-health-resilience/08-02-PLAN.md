```typescript
// GET /api/connectors/health-status — returns health status for all connectors
app.get("/api/connectors/health-status", isAuthenticated, async (req, res) => {
  try {
    const orgId = (req as any).user?.orgId;
    const healthStates = getConnectorHealthStatus();
    const circuitStates = getAllCircuitBreakerStates();

    // Get all connectors for this org to filter health data
    const { items: orgConnectors } = await storage.getConnectorsPaginated({ orgId, offset: 0, limit: 1000 });
    const orgConnectorIds = new Set(orgConnectors.map((c) => c.id));

    const result = orgConnectors.map((connector) => {
      const health = healthStates.get(connector.id);
      const circuit = circuitStates.get(connector.id);
      return {
        connectorId: connector.id,
        connectorName: connector.name,
        connectorType: connector.type,
        health: health
          ? {
              status: health.status,
              consecutiveFailures: health.consecutiveFailures,
              lastCheckAt: health.lastCheckAt ? new Date(health.lastCheckAt).toISOString() : null,
              lastSuccessAt: health.lastSuccessAt ? new Date(health.lastSuccessAt).toISOString() : null,
              latencyMs: health.latencyMs,
              restartCount: health.restartCount,
              nextRestartAt: health.nextRestartAt ? new Date(health.nextRestartAt).toISOString() : null,
            }
          : null,
        circuitBreaker: circuit
          ? {
              state: circuit.state,
              failures: circuit.failures,
              openUntil: circuit.openUntil ? new Date(circuit.openUntil).toISOString() : null,
            }
          : { state: "closed", failures: 0, openUntil: null },
      };
    });

    res.json(result);
  } catch (error) {
    logger.child("routes").error("Failed to fetch health status", { error: String(error) });
    res.status(500).json({ message: "Failed to fetch connector health status" });
  }
});

// POST /api/connectors/:id/circuit-breaker/reset — reset circuit breaker for a connector
app.post(
  "/api/connectors/:id/circuit-breaker/reset",
  isAuthenticated,
  resolveOrgContext,
  requireOrgId,
  requireMinRole("admin"),
  validatePathId("id"),
  async (req, res) => {
    try {
      const orgId = (req as any).orgId;
      const connector = await storage.getConnector(p(req.params.id));
      if (!connector || connector.orgId !== orgId) {
        return res.status(404).json({ message: "Connector not found" });
      }
      resetConnectorCircuitBreaker(connector.id);
      const state = getCircuitBreakerState(connector.id);
      res.json({ connectorId: connector.id, circuitBreaker: { state: state.state, failures: state.failures } });
    } catch (error) {
      logger.child("routes").error("Failed to reset circuit breaker", { error: String(error) });
      res.status(500).json({ message: "Failed to reset circuit breaker" });
    }
  },
);
```

IMPORTANT: The `health-status` route MUST be registered BEFORE any `/:id` routes to avoid Express matching "health-status" as a path parameter. Check placement carefully.
