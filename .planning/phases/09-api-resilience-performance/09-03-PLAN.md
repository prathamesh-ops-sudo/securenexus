<objective>
Convert triage to async 202-Accepted flow with job queue and add AI performance budgets.

Purpose: AI triage is a long-running operation that should not block the HTTP response (API-03). Converting to async processing with job queue gives retry, dedup, and dead-letter for free. Performance budgets for AI endpoints catch latency regressions (PERF-02 through PERF-05).

Output: Async triage endpoint (202 + job ID + poll URL), ai_triage job handler, SSE notification on completion, AI endpoint performance budgets, tests.
</objective>

       ai_triage: async (job) => {
         const { alertId, orgId } = job.payload;
         const { triageAlert, buildThreatIntelContext } = await import("./ai");
         const alert = await storage.getAlert(alertId);
         if (!alert) {
           return { error: "Alert not found", alertId };
         }
         const threatIntelCtx = await buildThreatIntelContext([alert]);
         const result = await triageAlert(alert, threatIntelCtx, orgId);

         // Broadcast completion via SSE
         const { broadcastEvent } = await import("./event-bus");
         broadcastEvent({
           type: "ai:triage_complete" as any, // EventType will include this after modification
           orgId: orgId || null,
           data: { jobId: job.id, alertId, result },
         });

         return { alertId, result };
       },

           "/api/ai/triage/jobs/:jobId",
           isAuthenticated,
           resolveOrgContext,
           async (req, res) => {
             const jobId = p(req.params.jobId);
             const job = await storage.getJob(jobId);
             if (!job) {
               return res.status(404).json({ message: "Job not found" });
             }
             // Verify org access
             const orgId = (req as any).orgId || (req as any).user?.orgId;
             if (job.orgId && orgId && job.orgId !== orgId) {
               return res.status(404).json({ message: "Job not found" });
             }
             if (job.status === "completed") {
               return res.json({ status: "completed", result: job.result });
             }
             if (job.status === "failed" || job.status === "dead_letter") {
               return res.json({ status: "failed", error: job.error });
             }
             res.json({ status: job.status }); // pending, processing
           },
         );
