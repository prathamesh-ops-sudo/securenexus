<objective>
Implement API key rotation with a 24-hour grace period on the old key.

Purpose: Allow consumers to rotate API keys without downtime. The old key remains valid for 24 hours after rotation, giving consumers time to update their configurations. This prevents the "hard cutover" problem where rotating a key immediately breaks all consumers.

Output: Schema additions for grace period tracking, rotation endpoint, updated auth middleware with grace period support, storage functions for deprecation.
</objective>

export function hashApiKey(key: string): string {
  return createHash("sha256").update(key).digest("hex");
}

export function generateApiKey(): { key: string; prefix: string; hash: string } {
  const key = `snx_${randomBytes(32).toString("hex")}`;
  const prefix = key.slice(0, 12);
  const hash = hashApiKey(key);
  return { key, prefix, hash };
}

export async function apiKeyAuth(req: Request, res: Response, next: NextFunction) {
  const header = req.headers["x-api-key"] || req.headers.authorization?.replace("Bearer ", "");

    return replyUnauthenticated(res, "Missing API key.", ERROR_CODES.API_KEY_MISSING);

    return replyUnauthenticated(res, "Invalid API key.", ERROR_CODES.API_KEY_INVALID);

1. Add POST /api/api-keys/:id/rotate endpoint to server/routes/ingestion.ts. Place it after the existing POST /api/api-keys route (around line 160). Follow the same middleware pattern as existing routes:

```typescript
app.post(
  "/api/api-keys/:id/rotate",
  isAuthenticated,
  resolveOrgContext,
  async (req, res) => {
    try {
      const { id } = req.params;
      const orgId = (req as any).orgId;

      // 1. Get the existing key
      const existingKey = await storage.getApiKeyById(id);
      if (!existingKey || existingKey.orgId !== orgId) {
        return replyNotFound(res, "API key not found.");
      }
      if (!existingKey.isActive) {
        return replyBadRequest(res, "Cannot rotate an inactive API key.");
      }

      // 2. Generate new key
      const { key, prefix, hash } = generateApiKey();

      // 3. Create new key record (inherits name, scopes, orgId from old key)
      const newKey = await storage.createApiKey({
        orgId: existingKey.orgId,
        name: `${existingKey.name} (rotated)`,
        keyHash: hash,
        keyPrefix: prefix,
        scopes: existingKey.scopes,
        createdBy: (req as any).user?.id || null,
      });

      // 4. Deprecate old key with 24h grace period
      await storage.deprecateApiKey(id, newKey.id);

      // 5. Audit log
      await storage.createAuditLog({
        userId: (req as any).user?.id,
        orgId,
        action: "api_key.rotated",
        resourceType: "api_key",
        resourceId: id,
        changes: {
          oldKeyId: id,
          oldKeyPrefix: existingKey.keyPrefix,
          newKeyId: newKey.id,
          newKeyPrefix: prefix,
          graceExpiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(),
        },
      });

      return reply(res, {
        newKey: {
          id: newKey.id,
          key, // shown only once
          prefix,
          scopes: existingKey.scopes,
        },
        oldKey: {
          id: existingKey.id,
          prefix: existingKey.keyPrefix,
          deprecatedAt: new Date().toISOString(),
          graceExpiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(),
        },
      }, {}, 200);
    } catch (err) {
      logger.child("ingestion").error("API key rotation failed", { error: String(err) });
      return replyError(res, "Failed to rotate API key.");
    }
  },
);
```
